<script>

(function () {

    /* =================================================
       CHECK ANALYSIS DATA
       ================================================= */

    if (typeof analysisData === "undefined") {

        alert(
            "analysis.js could not be loaded. " +
            "Make sure analysis.js is in the same " +
            "folder as dashboard.html."
        );

        return;
    }


    /* =================================================
       HELPER
       ================================================= */

    function setText(id, value) {

        var element =
            document.getElementById(id);

        if (element) {
            element.textContent = value;
        }
    }


    function percentage(value, total) {

        if (!total || total === 0)
            return 0;

        return (value / total) * 100;
    }


    function setBar(id, value, total) {

        var element =
            document.getElementById(id);

        if (!element)
            return;

        var percent =
            percentage(value, total);

        element.style.width =
            percent.toFixed(1) + "%";
    }


    /* =================================================
       MAIN COUNTERS
       ================================================= */

    setText(
        "totalPackets",
        Number(analysisData.totalPackets).toLocaleString()
    );


    setText(
        "tcpPackets",
        Number(analysisData.tcpPackets).toLocaleString()
    );


    setText(
        "udpPackets",
        Number(analysisData.udpPackets).toLocaleString()
    );


    setText(
        "dnsPackets",
        Number(analysisData.dnsPackets).toLocaleString()
    );


    /* =================================================
       TCP / UDP PERCENTAGES
       ================================================= */

    var total =
        Number(analysisData.totalPackets);


    var tcp =
        Number(analysisData.tcpPackets);


    var udp =
        Number(analysisData.udpPackets);


    setText(
        "tcpPercentage",
        percentage(tcp, total).toFixed(1)
        + "% of capture"
    );


    setText(
        "udpPercentage",
        percentage(udp, total).toFixed(1)
        + "% of capture"
    );


    /* =================================================
       PROTOCOL VALUES
       ================================================= */

    var protocols =
        analysisData.protocols;


    setText(
        "tcpProtocolValue",
        protocols.TCP
        + " • "
        + percentage(protocols.TCP, total).toFixed(1)
        + "%"
    );


    setText(
        "udpProtocolValue",
        protocols.UDP
        + " • "
        + percentage(protocols.UDP, total).toFixed(1)
        + "%"
    );


    setText(
        "tlsProtocolValue",
        protocols.TLS
        + " • "
        + percentage(protocols.TLS, total).toFixed(1)
        + "%"
    );


    setText(
        "dnsProtocolValue",
        protocols.DNS
        + " • "
        + percentage(protocols.DNS, total).toFixed(1)
        + "%"
    );


    setText(
        "httpProtocolValue",
        protocols.HTTP
        + " • "
        + percentage(protocols.HTTP, total).toFixed(1)
        + "%"
    );


    setText(
        "icmpv6ProtocolValue",
        protocols.ICMPv6
        + " • "
        + percentage(protocols.ICMPv6, total).toFixed(1)
        + "%"
    );


    /* =================================================
       PROTOCOL BARS
       ================================================= */

    setBar(
        "tcpBar",
        protocols.TCP,
        total
    );


    setBar(
        "udpBar",
        protocols.UDP,
        total
    );


    setBar(
        "tlsBar",
        protocols.TLS,
        total
    );


    setBar(
        "dnsBar",
        protocols.DNS,
        total
    );


    setBar(
        "httpBar",
        protocols.HTTP,
        total
    );


    setBar(
        "icmpv6Bar",
        protocols.ICMPv6,
        total
    );


    /* =================================================
       SECURITY
       ================================================= */

    setText(
        "retransmissions",
        analysisData.tcpRetransmissions
    );


    setText(
        "tcpResets",
        analysisData.tcpResets
    );


    setText(
        "securityIndicators",
        analysisData.securityIndicators
    );


    /* =================================================
       PORT SCAN STATUS
       ================================================= */

    var portScanElement =
        document.getElementById("portScan");


    if (!analysisData.tcpPortScanAvailable) {

        portScanElement.textContent =
            "UNAVAILABLE";

        portScanElement.className =
            "badge badge-info";

    }

    else if (analysisData.tcpPortScanDetected) {

        portScanElement.textContent =
            "DETECTED";

        portScanElement.className =
            "badge badge-alert";

    }

    else {

        portScanElement.textContent =
            "NONE";

        portScanElement.className =
            "badge badge-ok";

    }


    /* =================================================
       DNS STATUS
       ================================================= */

    var dnsAnomalyElement =
        document.getElementById("dnsAnomaly");


    if (analysisData.dnsAnomalyDetected) {

        dnsAnomalyElement.textContent =
            "DETECTED";

        dnsAnomalyElement.className =
            "badge badge-alert";

    }

    else {

        dnsAnomalyElement.textContent =
            "NONE";

        dnsAnomalyElement.className =
            "badge badge-ok";

    }


    /* =================================================
       GENERIC TABLE FUNCTION
       ================================================= */

    function fillTable(
        elementId,
        data,
        firstColumn
    ) {

        var tbody =
            document.getElementById(elementId);

        if (!tbody)
            return;


        tbody.innerHTML = "";


        if (!data || data.length === 0) {

            var emptyRow =
                document.createElement("tr");

            emptyRow.innerHTML =
                '<td colspan="3">No data available</td>';

            tbody.appendChild(emptyRow);

            return;
        }


        var i;

        for (i = 0;
             i < data.length;
             i++) {

            var row =
                document.createElement("tr");


            var rank =
                document.createElement("td");

            rank.textContent =
                i + 1;


            var name =
                document.createElement("td");

            name.textContent =
                data[i].name;


            var count =
                document.createElement("td");

            count.textContent =
                Number(data[i].count).toLocaleString();


            row.appendChild(rank);

            row.appendChild(name);

            row.appendChild(count);

            tbody.appendChild(row);
        }

    }


    /* =================================================
       SOURCE IP TABLE
       ================================================= */

    fillTable(
        "sourceTable",
        analysisData.topSources
    );


    /* =================================================
       DESTINATION IP TABLE
       ================================================= */

    fillTable(
        "destinationTable",
        analysisData.topDestinations
    );


    /* =================================================
       TCP PORT TABLE
       ================================================= */

    fillTable(
        "tcpPortTable",
        analysisData.topTCPPorts
    );


    /* =================================================
       UDP PORT TABLE
       ================================================= */

    fillTable(
        "udpPortTable",
        analysisData.topUDPPorts
    );


})();

</script>
